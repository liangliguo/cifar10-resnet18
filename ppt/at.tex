\section{背景与实验目标}

\begin{frame}{什么是对抗样本？}
  \begin{block}{核心现象}
    对于一张可被模型正确分类的图像，叠加一个\textbf{人眼几乎察觉不到}的微小扰动，
    模型可能以高置信度产生错误分类。这个被扰动后的图像称为\textbf{对抗样本}。
  \end{block}
  \vspace{0.5em}
  \begin{itemize}
    \item 扰动 $\br$ 幅度较小：$\norm{\br}$ 接近 0，视觉上与原图差异极小。
    \item 但模型输出 $\hat k(\bx+\br) \neq \hat k(\bx)$。
    \item 说明深度网络的决策边界在高维空间中可能距离正常样本较近。
  \end{itemize}
  \vspace{0.5em}
  \alert{研究意义：} 评估模型鲁棒性、理解决策边界、改进训练（对抗训练）。
\end{frame}

\begin{frame}{本次实验的三条主线}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{① 训练模型}
        \footnotesize
        在 CIFAR-10 上微调 ResNet-18，得到一个准确的\textbf{基线模型}，作为被攻击对象。
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{② DeepFool 攻击}
        \footnotesize
        \textbf{（核心）}用最小扰动诱导基线模型产生误分类，并与 FGSM / APGD 对比。
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{③ 对抗训练}
        \footnotesize
        用 PGD 对抗训练得到\textbf{鲁棒模型}，系统评估防御策略的有效性。
      \end{block}
    \end{column}
  \end{columns}

\section{模型训练}

\begin{frame}{数据集与预处理}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{CIFAR-10}：10 类自然图像，原生 $32\times32$。
      \begin{itemize}
        \item 训练 / 验证 / 测试 = \textbf{45000 / 5000 / 10000}
        \item 数据增广：\texttt{RandomCrop(32, padding=4)} + \texttt{RandomHorizontalFlip}
        \item 归一化：均值 $[0.485,0.456,0.406]$，标准差 $[0.229,0.224,0.225]$
        \item 攻击阶段图像保持在 $[0,1]$ 像素空间，\textbf{归一化集成到模型}内部
              （避免攻击库的 \texttt{clamp(0,1)} 破坏对抗样本）。
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{归一化集成到模型中的原因}
        \footnotesize
        攻击需要在合法像素区间 $[0,1]$ 内裁剪。若先归一化再攻击，
        \texttt{clamp(0,1)} 会截断归一化后的负值，破坏图像表示。
        统一约定：\textbf{数据 $[0,1]$，归一化作为模型第一层}。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{模型结构：CIFAR 版 ResNet-18}
  直接使用 ImageNet 版 ResNet-18 会在网络初始阶段将 $32\times32$ 下采样至 $8\times8$，
  造成较明显的空间信息损失。因此需做两处\textbf{适配小图}的改造：
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item \texttt{conv1}：$7\times7$ stride 2 $\;\to\;$ \textbf{$3\times3$ stride 1, padding 1}
        \item \texttt{maxpool}：$\;\to\;$ \textbf{Identity（去掉）}
        \item \texttt{fc}：输出维度改为 \textbf{10 类}
      \end{itemize}
      \vspace{0.4em}
      \begin{itemize}
        \item 参数量 $\approx 11.2$\,M，从零开始训练。
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{为什么不用预训练 ResNet-18}
        \footnotesize
        \begin{itemize}
          \item \textbf{stem 已调整}：预训练的 $7\times7$ \texttt{conv1} + \texttt{maxpool} 是为 $224\times224$ 大图设计的，
                将其替换为 $3\times3$、去掉 maxpool 后，\textbf{权重张量形状不匹配}，stem 无法直接加载。
          \item \textbf{迁移收益有限}：ImageNet 的高分辨率纹理特征对 $32\times32$ 小图收益有限。
          \item \textbf{无对抗训练基线}：从零训练得到\textbf{纯 CIFAR} 模型，避免迁移学习的混杂因素干扰后续\textbf{对抗鲁棒性}结论；4.5 万训练样本也足以从零训练至约 $88\sim89\%$。
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{DeepFool 攻击方法}
% ================================================================

\begin{frame}{对抗攻击的形式化定义}
  对于分类器 $f$ 与样本 $\bx$，记其预测标签
  \[
    \hat k(\bx) = \argmax_k f_k(\bx).
  \]
  寻找“最小”对抗扰动可写成优化问题：
  \[
    \Delta(\bx;f) \;:=\; \min_{\br}\; \norm{\br}_2
    \quad \text{s.t.}\quad \hat k(\bx+\br) \neq \hat k(\bx).
  \]
  \begin{itemize}
    \item $\Delta(\bx;f)$ 度量了样本 $\bx$ 到\textbf{决策边界}的距离。
    \item 该距离越小，表示模型在该样本附近的鲁棒性越低。
    \item \textbf{DeepFool 的目标：} 高效、准确地求解这个最小扰动。
  \end{itemize}
\end{frame}

\begin{frame}{回顾 FGSM 及其局限}
  经典攻击方法 FGSM（Fast Gradient Sign Method）执行单步更新：
  \[
    \br = \varepsilon \cdot \sign\!\big(\nabla_{\bx} J(\theta,\bx,y)\big).
  \]
  \begin{itemize}
    \item \textbf{优点：} 单步计算、效率较高，方向解释明确（沿损失上升最快方向更新）。
    \item \textbf{局限一：} 取 $\sign$ 会将每个像素维度的扰动推至 $\pm\varepsilon$，扰动\textbf{较粗糙且偏大}。
    \item \textbf{局限二：} 不是为“最小扰动”设计的，$\varepsilon$ 需要\textbf{人工设定}。
  \end{itemize}
  \vspace{0.4em}
  \begin{exampleblock}{由此引出 DeepFool}
    能否减少人工参数调节，自动找到\textbf{恰好跨越决策边界}的最小扰动？
  \end{exampleblock}
\end{frame}

\begin{frame}{几何直觉：决策边界的局部超平面近似}
  \small
  \begin{itemize}
    \item 在样本附近，把分类器\textbf{局部线性化}，决策边界近似为一个个\textbf{超平面}。
    \item 若要使样本改变类别，扰动范数最小的方向通常对应于到\textbf{最近边界}的\textbf{垂直方向}。
    \item 这对应于“点到平面的最短距离”这一线性代数经典问题。
  \end{itemize}
  \vspace{0.1em}
  \centering
  \begin{tikzpicture}[>=Stealth, scale=0.82]
    % 最近边界（较陡的线，离 x 近）
    \draw[thick,gray] (1.7,-1.6) -- (2.1,1.8);
    % 较远边界（较平的线，离 x 远）
    \draw[thick,gray] (-1.8,2.7) -- (3.8,3.1);
    % 样本点 x
    \fill (0,0) circle (2.5pt);
    \node[below left] at (0,0) {$\bx$};
    % 选中的最小扰动：指向最近边界
    \draw[->,very thick,orange!85!red] (0,0) -- (1.97,0.03)
      node[midway,above] {\small $\br_*$（最小）};
    % 被舍弃的更远方向
    \draw[->,thick,gray,dashed] (0,0) -- (0,2.92)
      node[midway,right] {\small 更远 → 舍弃};
    % 区域标注
    \node at (-1.25,-1.1) {\small class A（当前）};
    \node at (3.0,-0.7) {\small class B};
    \node at (-1.1,3.4) {\small class C};
  \end{tikzpicture}
  \par\vspace{0.1em}
  \scriptsize 在所有相邻类别的边界中，DeepFool 选择\textbf{最近}的边界，并进行垂直投影。
\end{frame}

\begin{frame}{二分类：线性分类器（基本情形）}
  设二分类器 $f(\bx)=\bw^\top\bx + b$，预测 $\hat k(\bx)=\sign\!\big(f(\bx)\big)$。
  \vspace{0.4em}

  样本 $\bx$ 到超平面 $\{\,\bx: f(\bx)=0\,\}$ 的最小扰动可由\textbf{正交投影}得到：
  \[
    \boxed{\;\br_*(\bx) = -\,\frac{f(\bx)}{\norm{\bw}_2^{2}}\,\bw\;}
  \]
  \begin{itemize}
    \item 方向：沿法向量 $\bw$（边界的垂直方向）。
    \item 大小：$\norm{\br_*}_2 = \dfrac{|f(\bx)|}{\norm{\bw}_2}$，对应于\textbf{点到平面的距离}。
    \item 这是可直接计算的闭式解。
  \end{itemize}
\end{frame}

\begin{frame}{二分类：一般（非线性）分类器}
  实际神经网络通常是非线性的，没有全局超平面。DeepFool 的做法是：\textbf{反复线性化 + 迭代}。
  \vspace{0.4em}

  在当前迭代点 $\bx_i$ 把 $f$ 一阶展开：
  \[
    f(\bx_i + \br) \approx f(\bx_i) + \nabla f(\bx_i)^\top \br .
  \]
  令其等于 0，得到当前迭代步的最小扰动：
  \[
    \br_i = -\,\frac{f(\bx_i)}{\norm{\nabla f(\bx_i)}_2^{2}}\,\nabla f(\bx_i),
    \qquad \bx_{i+1} = \bx_i + \br_i .
  \]
  \begin{itemize}
    \item 每次迭代都重新计算梯度（重新线性化），逐步逼近真实弯曲的边界。
    \item 直到 $\hat k(\bx_i)$ 改变为止，通常只需 \textbf{2--4 步}。
  \end{itemize}
\end{frame}

\begin{frame}{多分类：在所有边界中选最近的一条}
  对 $c$ 类分类器 $f:\mathbb{R}^n\to\mathbb{R}^c$，记原始标签 $\hat k(\bx_0)$。
  在迭代点 $\bx_i$，对每个其他类别 $k \neq \hat k(\bx_0)$ 定义：
  \[
    \bw'_k = \nabla f_k(\bx_i) - \nabla f_{\hat k(\bx_0)}(\bx_i),
    \qquad
    f'_k = f_k(\bx_i) - f_{\hat k(\bx_0)}(\bx_i).
  \]
  \begin{itemize}
    \item $\bw'_k$：类别 $k$ 与当前类之间“等分边界”的法向。
    \item $f'_k$：当前点相对该边界的“带符号距离量”。
  \end{itemize}
  \vspace{0.3em}
  选出\textbf{最近}的边界：
  \[
    \hat l = \argmin_{k \neq \hat k(\bx_0)} \frac{|f'_k|}{\norm{\bw'_k}_2}.
  \]
\end{frame}

\begin{frame}{多分类：最小扰动公式}
  朝最近边界 $\hat l$ 做正交投影，得到本步扰动：
  \[
    \boxed{\;
      \br_i = \frac{|f'_{\hat l}|}{\norm{\bw'_{\hat l}}_2^{2}}\,\bw'_{\hat l}
    \;}
  \]
  随后更新 $\bx_{i+1} = \bx_i + \br_i$，重复该过程直到类别改变。
  \vspace{0.6em}

  最终累计扰动并做\textbf{过冲}（overshoot），确保确实越过边界：
  \[
    \hat\br = (1+\eta)\sum_i \br_i,
    \qquad \eta = 0.02 .
  \]
  \footnotesize 过冲系数 $1.02$ 使扰动略微越过边界，避免样本停留在边界附近，并因数值误差或量化影响仍被判为原类别。
\end{frame}

\begin{frame}{算法伪代码（多分类 DeepFool）}
  \begin{block}{}
  \footnotesize
  \begin{enumerate}
    \item 初始化 $\bx_0 \gets \bx$，$\ i \gets 0$
    \item \textbf{while} $\hat k(\bx_i) = \hat k(\bx_0)$ \textbf{do}
    \item \quad \textbf{for} $k \neq \hat k(\bx_0)$:
          \ $\bw'_k \gets \nabla f_k(\bx_i) - \nabla f_{\hat k(\bx_0)}(\bx_i)$,
          \ $f'_k \gets f_k(\bx_i) - f_{\hat k(\bx_0)}(\bx_i)$
    \item \quad 选择最近边界 \ $\hat l \gets \argmin\limits_{k \neq \hat k(\bx_0)} \dfrac{|f'_k|}{\norm{\bw'_k}_2}$
    \item \quad $\br_i \gets \dfrac{|f'_{\hat l}|}{\norm{\bw'_{\hat l}}_2^{2}}\,\bw'_{\hat l}$,
          \ \ $\bx_{i+1} \gets \bx_i + \br_i$,
          \ \ $i \gets i+1$
    \item \textbf{end while}
    \item \textbf{return} \ $\hat\br = (1+\eta)\sum_i \br_i$
  \end{enumerate}
  \end{block}
  \footnotesize 复杂度主要来自每步对 $c$ 个类别求梯度；在 CIFAR-10（$c=10$）规模下计算开销可控。
  本实验用 \texttt{torchattacks.DeepFool(steps=20, overshoot=0.02)}。
\end{frame}

\section{攻击实验与结果}
% ================================================================

\begin{frame}{鲁棒性度量与实验设置}
  最小扰动攻击的核心指标不是“成功率”，而是\textbf{平均/中位 $L_2$ 扰动}——数值越小，表示模型鲁棒性越低。
  论文用\textbf{平均相对扰动}：
  \[
    \hat\rho_{adv}(f) = \frac{1}{|\mathcal{D}|}
      \sum_{\bx \in \mathcal{D}} \frac{\norm{\hat\br(\bx)}_2}{\norm{\bx}_2}.
  \]
  \begin{itemize}
    \item \textbf{被攻击模型：} 上一节训练得到的 ResNet-18（干净准确率 88.53\%）。
    \item \textbf{评测样本：} FGSM / APGD 的指标（Acc / 成功率 / $L_2$ / $\hat\rho_{adv}$）用\textbf{全量测试集 10000 张}；
          DeepFool 计算成本较高，用 \textbf{1000 张}；\textbf{可视化}（分布用 1000、t-SNE 用 200、混淆矩阵）口径见各图注。
    \item \textbf{对照攻击：} FGSM、APGD（均 $L_\infty$，$\varepsilon=8/255$）。
    \item \textbf{成功率口径：} \textbf{条件攻击成功率} = 干净样本上分类正确、经攻击后变为误分类的比例。
  \end{itemize}
\end{frame}

\begin{frame}{实测结果：三种攻击对比（普通模型，白盒）}

  \footnotesize
  \begin{tabular}{lccccc}

  \end{itemize}
\end{frame}

\section{对抗训练（防御）}
% ================================================================

\begin{frame}{对抗训练：使用最坏情形样本训练模型}
  防御方法：训练时\textbf{先用 PGD 为每批干净样本在线生成对抗样本}，再用其（标签仍为原始 $y$）训练，
  促使模型在受攻击条件下仍保持正确分类。形式化为\textbf{min-max（鞍点）问题}：
  \[
    \min_{\theta}\ \mathbb{E}_{(\bx,y)}\Big[\ \max_{\norm{\bdelta}_\infty \le \varepsilon}\ L\big(f_\theta(\bx+\bdelta),\,y\big)\Big].
  \]
  \begin{itemize}
    \item \textbf{内层 $\max$：} 在预算 $\varepsilon$ 内寻找使损失最大的扰动（用 PGD 近似）。
    \item \textbf{外层 $\min$：} 更新参数 $\theta$，降低“最坏情况”损失，从而提升鲁棒性。
    \item 对抗样本\textbf{在线生成}：模型参数持续更新，因此需基于当前模型重新执行 PGD。
  \end{itemize}
\end{frame}

\begin{frame}{PGD：求解内层 $\max$ 的迭代攻击}
  PGD（Projected Gradient Descent）= \textbf{多步 FGSM + 投影}，
  用来近似内层 $\max$——在 $\varepsilon$ 预算内寻找使损失最大的“最坏扰动”：
  \[
    \bx^{t+1} = \Pi_{\mathcal{B}_\varepsilon(\bx)}\!\Big(\bx^{t} + \alpha\,\sign\!\big(\nabla_{\bx} L(\theta,\bx^{t},y)\big)\Big),
    \qquad t = 0,1,\dots,K{-}1.
  \]
  \begin{itemize}
    \item \textbf{随机起点：} $\bx^{0} = \bx + \mathrm{Uniform}(-\varepsilon,\varepsilon)$，避免每次从同一点出发、削弱梯度混淆。
    \item \textbf{投影 $\Pi$：} 每步把结果裁回 $L_\infty$ 球（$\norm{\bx^{t}-\bx}_\infty \le \varepsilon$）并保持 $\bx^{t}\in[0,1]$。
    \item \textbf{步长 $\alpha$、步数 $K$：} 训练用 \textbf{PGD-7}（$\alpha = 2.5\varepsilon/K \approx 0.36\varepsilon$）；
          评估用更强的 \textbf{PGD-20}（$\alpha = 2/255$，随机重启）。
  \end{itemize}
  \vspace{0.2em}
  \footnotesize 与 FGSM（单步将扰动推至 $\pm\varepsilon$）相比，PGD 在预算内\textbf{迭代增加损失}，
  生成的高损失对抗样本攻击强度更高，符合对抗训练内层 $\max$ 的目标。
\end{frame}

\begin{frame}{为什么用 PGD，而不是 DeepFool？}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{PGD（用于训练）}
        \footnotesize
        在\textbf{固定预算 $\varepsilon$ 内最大化损失} → 提供“最坏情况”样本，
        与内层 $\max L$ 的优化目标一致。\\[0.3em]
        \emph{因此适合用于提升固定扰动预算下的鲁棒性。}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{DeepFool（不用于训练）}
        \footnotesize
        求\textbf{最小扰动、达到类别改变即停止} → 样本通常位于边界附近，攻击强度有限，
        且每个样本扰动大小不一、\textbf{没有统一预算}。\\[0.3em]
        \emph{因此不适合作为固定预算对抗训练的内层攻击。}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{exampleblock}{结论}
    \footnotesize
    \textbf{PGD} 同时用于训练与评估；\textbf{DeepFool} 是\textbf{评估}工具
    （用于测量最小扰动大小，并估计样本到决策边界的距离）。
  \end{exampleblock}
\end{frame}

\begin{frame}{PGD 对抗训练实验设置}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \footnotesize
      \begin{itemize}
        \item 训练集：全量 50000，原生 $32\times32$ + 增广
        \item 内层攻击：\textbf{PGD-7}，$L_\infty$，$\varepsilon = 8/255$
        \item $\varepsilon$ 预热：前 5 轮从 $0.5\varepsilon$ 线性升到 $\varepsilon$
        \item 损失：$(1-\lambda)\,\mathrm{CE}_{\text{clean}} + \lambda\,\mathrm{CE}_{\text{adv}}$，$\lambda=0.5$
              （保留干净样本性能，\textbf{防止塌缩}）
        \item 优化器：\texttt{SGD}（lr $=0.1$, mom $=0.9$, wd $=5\times10^{-4}$）
        \item 调度：\texttt{MultiStepLR} 在 40 / 60 轮 $\div 10$，共 \textbf{80} 轮
        \item 梯度裁剪：上限 $1.0$
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{鲁棒性评估控制项}
        \footnotesize
        \begin{itemize}
          \item 干净/对抗损失\textbf{各半}，避免模型塌缩（Clean/Robust 双低）。
          \item 评估用\textbf{强攻击}（APGD），并按强度阶梯 FGSM < PGD-20 < APGD 排查梯度混淆。
          \item Clean 与 Robust Acc \textbf{配对报告}。
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

    \bottomrule
  \end{tabular}
  \end{table}
  \vspace{0.4em}
  \footnotesize

  代价是 Clean Acc 会下降（鲁棒性—准确率权衡）。\textbf{本实验中鲁棒模型 Clean 下降 2.3 个百分点（88.53\%→86.25\%），同时三种攻击下 Robust Acc 由近 0 提升至 45--56\%，未见塌缩；}且同一模型沿 FGSM > PGD-20 > APGD 依次递减，无梯度混淆迹象。

\section{讨论与总结}
% ================================================================

\begin{frame}{DeepFool 的优点与局限}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{优点}
        \begin{itemize}
          \item 扰动\textbf{最小}、几乎不可见
          \item \textbf{无需人工调节}关键超参数
          \item 迭代次数少、\textbf{计算效率较高}
          \item 几何直觉清晰、\textbf{便于解释}
          \item 提供可比的鲁棒性度量
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{局限}
        \begin{itemize}
          \item 默认\textbf{无定向}（不能指定目标类）
          \item 扰动位于边界附近、\textbf{对预处理敏感}（易被量化/压缩抵消，靠过冲缓解）
          \item 纯攻击\textbf{强度}略逊于 C\&W
          \item \textbf{不适合}用于对抗训练
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{不同攻击方法的对比}
  \begin{table}
  \centering
  \footnotesize
  \begin{tabular}{lll}
    \toprule
    方法 & 核心机制 & 特点 \\
    \midrule
    FGSM     & 沿梯度符号执行单步更新     & 单步、$L_\infty$、扰动较大 \\
    PGD      & 在 $\varepsilon$ 球内迭代最大化损失 & 迭代、$L_\infty$、用于训练与评估 \\
    APGD     & 自适应步长 + 重启     & 高强度评估基线（AutoAttack 组件） \\
    \alert{DeepFool} & \alert{投影到最近决策边界} & \alert{迭代、最小扰动、几何解释明确} \\
    C\&W     & 优化最小 $L_2$ 扰动   & 攻击强度高、扰动极小、计算成本较高 \\
    \bottomrule
  \end{tabular}
  \end{table}
  \vspace{0.5em}
  \begin{itemize}
    \item DeepFool 与 C\&W 属于“\textbf{最小扰动}”评估范式（估计到决策边界的距离）。
    \item FGSM / PGD / APGD 属于“\textbf{固定预算}”评估范式（分析给定 $\varepsilon$ 内的攻击效果）。
  \end{itemize}
\end{frame}

\begin{frame}{总结}
  \begin{enumerate}
    \item \textbf{训练模型：} CIFAR 版 ResNet-18（conv1 改 $3\times3$、去 maxpool），
          测试准确率 \textbf{88.53\%}，作为基线。
    \item \textbf{DeepFool 攻击：} 局部线性化、投影到最近决策边界，求\textbf{最小 $L_2$ 扰动}；
          实测平均 $L_2$ 仅 \textbf{0.16}，约 FGSM 的 $1/11$，攻击成功率相近。
    \item \textbf{对抗训练：} 用 PGD（非 DeepFool）做 min-max 训练得到鲁棒模型；
          可靠评估需 Clean/Robust 配对 + 强攻击（APGD）。
  \end{enumerate}
  \vspace{0.5em}
  \begin{block}{结论概括}
    DeepFool 可量化模型到决策边界的距离，PGD 对抗训练可提升固定扰动预算下的鲁棒性；
    鲁棒性评估必须采用强攻击与配对指标，以避免高估防御效果。
  \end{block}
\end{frame}
